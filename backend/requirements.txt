# Core web framework
# Relaxed to >= so pip can resolve a compatible set with pydantic>=2.10 on Python 3.13
fastapi>=0.111.0

# ASGI server to run FastAPI
uvicorn[standard]>=0.30.1

# ORM and database toolkit
sqlalchemy>=2.0.31

# Database migration tool
alembic>=1.13.2

# PostgreSQL adapter
# psycopg2-binary 2.9.9 has no Python 3.13 wheel — it tries to build from source
# and fails unless PostgreSQL dev tools (pg_config) are installed on the system.
# Using >=2.9.10 which ships pre-built wheels for Python 3.13 on macOS.
psycopg2-binary>=2.9.10

# Data validation and serialisation

# pydantic 2.10+ uses pydantic-core 2.27+ with PyO3 0.22+ which supports Python 3.13.
pydantic>=2.10.0

# Pydantic extension for loading settings from .env files
# 2.6.0+ is compatible with pydantic 2.10+
pydantic-settings>=2.6.0

# Loads .env files into os.environ at startup
